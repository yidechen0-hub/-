void quickSort(vector<int>& nums,int left,int right){
        if(left>=right)return;
        int l = left-1;
        int r = right+1;
        int tmp = nums[left];
        while(l<r){
            do l++;while(nums[l]<tmp);
            do r--;while(nums[r]>tmp);
            if(l<r){
                swap(nums[l],nums[r]);
            }
        }
        quickSort(nums,left,r);
        quickSort(nums,r+1,right);

    }

void maxHeapify(vector<int>& a, int i, int heapSize) {
        int l = i * 2 + 1, r = i * 2 + 2, largest = i;
        if (l < heapSize && a[l] > a[largest]) {
            largest = l;
        } 
        if (r < heapSize && a[r] > a[largest]) {
            largest = r;
        }
        if (largest != i) {
            swap(a[i], a[largest]);
            maxHeapify(a, largest, heapSize);
        }
    }
void buildMaxHeap(vector<int>& a, int heapSize) {
        for (int i = heapSize / 2; i >= 0; --i) {
            maxHeapify(a, i, heapSize);
        } 
    }

# 归并排序

~~~cpp
void Merge(int a[], int left, int mid, int right){
    int temp[right - left + 1];                   //临时数组用于存储排序时的数
    int i = left;                                 //分成两块 i指向左边的数字 j指向右边的数字 
    int j = mid + 1;
    int k = 0;                                    //k用于存储数字到临时数组

    while( i <= mid && j <= right ){
    	if(a[i] < a[j])    	                  //永远都是 i 和 j 指向的数进行比较
    	    temp[k++] = a[i++];                   //谁小，谁就先放到临时数组中
    	else
    	    temp[k++] = a[j++];
    }

    while( i <= mid )                             //如果左边还有数没放上去，就依次放上去
    	temp[k++] = a[i++];
    while( j <= right )                           //如果是右边还有同上
    	temp[k++] = a[j++];
    
    for(int m = left, n = 0; m <= right; m++, n++)//读取临时数组中的数
    	a[m] = temp[n];
}


void Merge_Sort(int a[], int left, int right){
    if( left == right )
    	return;

    int mid = (left + right)/2;
    //递归拆分成较小规模子序列排序 
    Merge_Sort(a, left, mid);            
    Merge_Sort(a, mid + 1, right);
    Merge(a, left, mid, right);      //合并较小规模问题解
}

~~~

void Bubbling_sorting(vector<int>&nums){
    int n = nums.size();
    bool flag = false;
    for(int i=n-1;i>=0;i--){
        flag = false;
        for(int j=0;j<i;j++){
            if(nums[j]>nums[j+1]){
                swap(nums[j+1],nums[j]);
                flag = true;
            }
        }
        if(flag == false){

        }
    }

# 计数排序

[2785. 将字符串中的元音字母排序 - 力扣（LeetCode）](https://leetcode.cn/problems/sort-vowels-in-a-string/solutions/3766881/jiang-zi-fu-chuan-zhong-de-yuan-yin-zi-m-qllf/?envType=daily-question&envId=2025-09-11)

计数排序（Counting Sort）是一种**非比较式**的整数排序算法。它通过统计数组中每个元素出现的次数来对数据进行排序，特别适用于**一定范围内**的整数排序。

### 核心思想

1.  **统计频次**：创建一个额外的“计数数组”（也称为桶），其大小等于待排序数据的**值域范围**。遍历原数组，统计每个元素出现的次数，并将这些次数存储在计数数组的对应位置。
2.  **累加计数 (可选，用于稳定排序)**：对计数数组进行累加操作，使其从“频次数组”转变为“位置数组”。此时，计数数组中的每个值表示小于或等于该索引值的元素个数，这有助于确定每个元素在最终排序数组中的正确位置。
3.  **构建结果**：根据计数数组（通常是累加后的），将原数组中的元素按顺序放置到一个新的输出数组中。

### 算法步骤（以升序排列为例）

假设待排序数组为 `arr`，长度为 `n`，元素的最小值为 `min_val`，最大值为 `max_val`。

1.  找出 `arr` 中的最大值 (`max_val`) 和最小值 (`min_val`)，确定值域范围 `range = max_val - min_val + 1`。
2.  创建一个大小为 `range` 的计数数组 `count`，初始化所有元素为 0。
3.  遍历 `arr`，对于每个元素 `arr[i]`，执行 `count[arr[i] - min_val]++`。这里减去 `min_val` 是为了将值域映射到 `[0, range-1]`，节省空间。
4.  （可选，用于实现稳定排序）对 `count` 数组进行累加：`for i from 1 to range-1: count[i] += count[i-1]`。累加后，`count[i]` 表示值小于等于 `i + min_val` 的元素个数。
5.  创建一个大小为 `n` 的输出数组 `output`。
6.  **关键：为了保证稳定性，需要从原数组的末尾开始向前遍历**：
    *   从 `i = n-1` 开始递减到 `0`。
    *   对于 `arr[i]`，计算其在 `count` 数组中的索引 `idx = arr[i] - min_val`。
    *   将 `arr[i]` 放置到 `output` 数组中位置 `count[idx] - 1` 上。
    *   将 `count[idx]` 减 1。
7.  将 `output` 数组复制回 `arr` 数组。

### C++ 实现（稳定版本）

```cpp
#include <iostream>
#include <vector>
#include <algorithm> // for std::min_element, std::max_element

void countingSort(std::vector<int>& arr) {
    if (arr.empty()) return; // 处理空数组

    // 步骤1: 找到最大值和最小值
    int min_val = *std::min_element(arr.begin(), arr.end());
    int max_val = *std::max_element(arr.begin(), arr.end());
    int range = max_val - min_val + 1;

    // 步骤2: 创建计数数组并初始化为0
    std::vector<int> count(range, 0);

    // 步骤3: 统计每个元素的出现次数
    for (int num : arr) {
        count[num - min_val]++;
    }

    // 步骤4: 对计数数组进行累加 (转换为位置信息)
    for (int i = 1; i < range; ++i) {
        count[i] += count[i - 1];
    }

    // 步骤5: 创建输出数组
    std::vector<int> output(arr.size());

    // 步骤6: 从后往前遍历原数组，填充输出数组 (保证稳定性)
    for (int i = arr.size() - 1; i >= 0; --i) {
        // 计算当前元素在计数数组中的索引
        int idx = arr[i] - min_val;
        // 将元素放到 output 中正确的位置 (count[idx] - 1)
        output[count[idx] - 1] = arr[i];
        // 更新计数
        count[idx]--;
    }

    // 步骤7: 将结果复制回原数组
    arr = output;
}

// 测试
int main() {
    std::vector<int> arr = {4, 2, 2, 8, 3, 3, 1};
    std::cout << "Original array: ";
    for (int num : arr) std::cout << num << " ";
    std::cout << std::endl;

    countingSort(arr);

    std::cout << "Sorted array: ";
    for (int num : arr) std::cout << num << " ";
    std::cout << std::endl;

    return 0;
}
```

### 时间复杂度

*   **O(n + k)**
    *   `n` 是待排序数组的元素个数。
    *   `k` 是待排序数据的**值域范围**（即 `max_val - min_val + 1`）。
*   遍历原数组统计频次：O(n)
*   遍历计数数组进行累加：O(k)
*   遍历原数组构建输出：O(n)
*   总体是线性的，当 `k` 不大时，效率非常高。

### 空间复杂度

*   **O(k)**
*   主要消耗在大小为 `k` 的计数数组上。另外还需要 O(n) 的输出数组。所以总空间复杂度是 O(n + k)，但通常说主导项是 O(k)。

### 优缺点

**优点：**

1.  **速度快**：时间复杂度是线性的 O(n+k)，在 `k` 相对较小时，比任何基于比较的排序算法（如快速排序、归并排序的 O(n log n)）都要快。
2.  **稳定**：相等的元素在排序后相对位置不变（如果按照上述从后往前填充的方式实现）。
3.  **非比较式**：不依赖于元素间的比较操作。

**缺点：**

1.  **适用范围有限**：只能用于排序整数（或可以映射到整数的数据）。不能直接用于浮点数、字符串等。
2.  **空间浪费严重**：当待排序数据的值域 `k` 非常大时（例如，排序 100 个数字，但它们的范围是 1 到 10亿），需要创建一个巨大的计数数组，造成极大的空间浪费，效率反而很低。
3.  **需要额外空间**：需要 O(n+k) 的额外空间来存储计数数组和输出数组。

### 适用场景

*   待排序的元素是**整数**。
*   整数的**范围 `k` 相对较小**，并且与元素个数 `n` 相差不大。
*   对排序速度要求很高，且内存充足。
*   常作为**基数排序 (Radix Sort)** 的子程序。

**总结：** 计数排序是一种高效的线性时间排序算法，但它对数据的分布有严格的要求（整数、范围小）。在满足条件的情况下，它是极佳的选择；否则，应考虑使用其他排序算法。
